# Core dependencies for Damanat PMS Video Analytics
#
# torch + torchreid notes:
#   torchreid (vehicle re-ID model) needs torch + torchvision. We pull them
#   from PyTorch's CPU-only index — without --extra-index-url pip resolves
#   to the GPU build (~2GB) which we don't need on the inference boxes.
#   Order matters: numpy<2 first, then torch/torchvision, then torchreid.
--extra-index-url https://download.pytorch.org/whl/cpu

numpy<2.0.0               # torch/torchreid not yet compatible with NumPy 2.x
torch>=2.2,<2.5           # CPU-only build (see --extra-index-url above)
torchvision>=0.17,<0.20   # Pinned to match the supported torch range
torchreid==0.2.5          # Vehicle re-ID matcher (used by tracking_manager)

# Phase 1 / WS-A OpenVINO INT8 export pipeline for OSNet ReID.
# ``onnx`` is needed at export time only; the inference path uses OpenVINO
# Runtime directly. ``nncf`` performs post-training INT8 quantisation.
# Also used by WS-B (color) and WS-C (type) classifier export pipelines.
onnx>=1.16.0
nncf>=2.10.0

# Plate-OCR cross-check (Wave 1 / WS-D).
#
# PaddleOCR-mobile reads license plates from floor crops at the marginal
# ReID score band (see src.ocr.plate_ocr and Phase 2 gating). The detection
# + recognition models live under ~/.paddlex/official_models and are
# downloaded on first use:
#
#     PP-OCRv5_mobile_det        ~ 5 MB
#     en_PP-OCRv5_mobile_rec     ~ 8 MB
#     PP-LCNet_x1_0_textline_ori ~ 7 MB   (only when use_angle_cls=True)
#
# Total on-disk footprint: ~12-20 MB for the mobile pipeline (the older
# PaddleOCR v3/v4 server variants are ~80 MB; we pin to mobile in code).
# The Python wheel chain (paddleocr + paddlepaddle + paddlex) is ~500 MB
# installed — gate it behind a feature flag in resource-constrained envs.
#
# On Windows + paddlepaddle 3.x oneDNN can trip a NotImpl at predict time;
# the plugin sets FLAGS_use_mkldnn=0 by default to avoid it.
paddleocr>=2.7.0
paddlepaddle>=2.6.0

# Phase 3 / T3.2 — FAISS-CPU gallery index. Optional at runtime: when
# absent the GalleryIndex falls back to a pure-numpy brute-force search
# with identical semantics (slower above ~5000 sessions). Pinning at
# 1.7+ for IndexIVFFlat parity across platforms.
faiss-cpu>=1.7.0
